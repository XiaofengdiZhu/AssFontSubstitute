```python
from __future__ import annotations

import queue
from pathlib import Path
from types import SimpleNamespace

from PySide6.QtCore import QThread, Signal

from ass_font_substitute.gui.settings_store import Settings


class CoreWorker(QThread):
    """串行核心工作者：独占 Cache/LibraryIndex（core Cache 单连接线程约束）。
    submit(tag, fn, *args, **kw) 入队；fn(ctx, *args, **kw) 的返回值经
    result 信号回 UI 线程，异常经 failed 信号。任务严格按提交顺序执行。"""

    result = Signal(object, object)   # (tag, payload)
    failed = Signal(object, str)      # (tag, "ExcType: msg")

    def __init__(self, settings: Settings, parent=None):
        super().__init__(parent)
        self._settings = settings
        self._q: queue.Queue = queue.Queue()
        self._stop = False

    def run(self):
        from ass_font_substitute.core.similarity.cache import Cache
        from ass_font_substitute.gui.paths import default_cache_path
        p = self._settings.get("cache_path") or str(default_cache_path())
        ctx = SimpleNamespace(settings=self._settings, cache=Cache(Path(p)),
                              snapshot=None, lib=None, lib_dirs=None)
        while not self._stop:
            item = self._q.get()
            if item is None:
                break
            tag, fn, args, kw = item
            try:
                self.result.emit(tag, fn(ctx, *args, **kw))
            except Exception as e:  # noqa: BLE001  单任务失败不拖垮工作者
                self.failed.emit(tag, f"{type(e).__name__}: {e}")

    def submit(self, tag, fn, *args, **kw):
        self._q.put((tag, fn, args, kw))

    def shutdown(self):
        self._stop = True
        self._q.put(None)
```

```python
        self.worker = CoreWorker(self.settings)
        self.worker.start()

        self.worker.shutdown()
        self.worker.wait(2000)
        super().closeEvent(e)
```

`src/ass_font_substitute/gui/settings_store.py`：

```python
from __future__ import annotations

import json
import os
import re
from pathlib import Path

from ass_font_substitute.core.coverage.matcher import COV_THRESHOLD
from ass_font_substitute.core.similarity.search import MIN_SIM, SAMPLE_TEXT, SIZE_RANGE

# 界面 stepper 边界（spec §10）：覆盖率 90–100 步长 1；相似率 50–100 步长 5；
# 大小区间 50–200 步长 1。存储一律小数，页面按百分数换算
COV_BOUNDS = (0.90, 1.00, 0.01)
SIM_BOUNDS = (0.50, 1.00, 0.05)
SIZE_BOUNDS = (0.50, 2.00, 0.01)

DEFAULTS: dict = {
    "font_dirs": [],
    "cov_threshold": COV_THRESHOLD,
    "min_sim": MIN_SIM,
    "size_min": SIZE_RANGE[0],
    "size_max": SIZE_RANGE[1],
    "n_pause": 5,
    "max_threads": max(1, os.cpu_count() or 4),   # 挂账：并行解析落地前仅存储
    "save_suffix": ".substituted.ass",
    "overwrite_default": False,
    "sample_text": SAMPLE_TEXT,
    "cache_path": "",                              # 空 = paths.default_cache_path()
}

_BOOL_KEYS = {"overwrite_default"}
_LIST_KEYS = {"font_dirs"}
_INT_KEYS = {"n_pause", "max_threads"}
_FLOAT_KEYS = {"cov_threshold", "min_sim", "size_min", "size_max"}
_SUFFIX_RE = re.compile(r"^\..+$")


class Settings:
    """JSON 设置。get/set 同步落盘（tmp + os.replace 原子写）；
    损坏/含未知键/类型不符的存量数据按默认处理，绝不崩 GUI。"""

    def __init__(self, path: Path | None = None):
        self.path = Path(path) if path is not None else None
        self.reload()

    def _file(self) -> Path:
        if self.path is not None:
            return self.path
        from ass_font_substitute.gui.paths import settings_path
        return settings_path()

    def reload(self) -> None:
        self._data = dict(DEFAULTS)
        f = self._file()
        if not f.exists():
            return
        try:
            stored = json.loads(f.read_text("utf-8"))
        except (OSError, json.JSONDecodeError):
            return
        for k, v in stored.items():
            if k in _BOOL_KEYS and isinstance(v, bool):
                self._data[k] = v
            elif k in _LIST_KEYS and isinstance(v, list):
                self._data[k] = [str(x) for x in v]
            elif k in _INT_KEYS and isinstance(v, int) and not isinstance(v, bool):
                self._data[k] = v
            elif k in _FLOAT_KEYS and isinstance(v, (int, float)) \
                    and not isinstance(v, bool):
                self._data[k] = float(v)
            elif k == "sample_text" and isinstance(v, str) and v.strip():
                self._data[k] = v
            elif k in ("save_suffix", "cache_path") and isinstance(v, str):
                self._data[k] = v

    def get(self, key: str):
        return self._data[key]

    def set(self, key: str, value) -> None:
        if key == "sample_text" and (not isinstance(value, str) or not value.strip()):
            raise ValueError("对比文字不能为空")
        if key == "save_suffix" and not (isinstance(value, str) and _SUFFIX_RE.match(value)):
            raise ValueError("另存后缀必须以点开头且非空，如 .substituted.ass")
        self._data[key] = value
        self.save()

    def save(self) -> None:
        f = self._file()
        f.parent.mkdir(parents=True, exist_ok=True)
        tmp = f.with_name(f.name + ".tmp")
        tmp.write_text(json.dumps(self._data, ensure_ascii=False, indent=2), "utf-8")
        os.replace(tmp, f)
```

```python
SEARCH_CHUNK = 8          # pull_chunk 每批候选数（UI 渐进填充节奏）


@dataclass
class DirStatus:
    path: str
    state: str                  # ok / stale / missing / offline
    added: int = 0
    changed: int = 0
    indexed_files: int = 0


def probe_dir_status(dir_path: str) -> DirStatus:
    """只读探测（URI mode=ro，绝不创建/写库）。missing=目录不存在；
    offline=库存在但打不开（只读盘/权限/损坏）；stale=有未入索引文件。"""
    d = Path(dir_path)
    if not d.is_dir():
        return DirStatus(dir_path, "missing")
    dbf = d / "fontindex.db"
    if not dbf.exists():
        st = DirStatus(dir_path, "stale")
        st.added = len(scan_dir(d))
        return st
    try:
        conn = sqlite3.connect(dbf.resolve().as_uri() + "?mode=ro", uri=True)
        known = {r[0]: (r[1], r[2]) for r in
                 conn.execute("SELECT rel_path, size, mtime_ns FROM files")}
        bad = {r[0]: (r[1], r[2]) for r in
               conn.execute("SELECT rel_path, size, mtime_ns FROM bad")}
        conn.close()
    except sqlite3.Error:
        return DirStatus(dir_path, "offline")
    st = DirStatus(dir_path, "ok", indexed_files=len(known))
    for p in scan_dir(d):
        rel = p.relative_to(d).as_posix()
        s = p.stat()
        cur = (s.st_size, s.st_mtime_ns)
        if rel in known:
            if known[rel] != cur:
                st.changed += 1
        elif rel in bad:
            if bad[rel] != cur:
                st.changed += 1
        else:
            st.added += 1
    if st.added or st.changed:
        st.state = "stale"
    return st


@dataclass
class LibrarySnapshot:
    lib: LibraryIndex
    offline_dirs: list[str] = field(default_factory=list)


def load_library(dirs: list[str]) -> LibrarySnapshot:
    """聚合打开库；打不开的目录（离线/只读）剔除并报告，不拖垮整体。"""
    usable, offline = [], []
    for d in dirs:
        try:
            open_db(Path(d) / "fontindex.db").close()
            usable.append(Path(d))
        except sqlite3.Error:
            offline.append(d)
    return LibrarySnapshot(lib=LibraryIndex.open(usable), offline_dirs=offline)


def ensure_library(ctx, dirs: list[str]) -> LibrarySnapshot:
    """聚合库按需（重）建：目录集不变则复用 ctx 上缓存。"""
    key = tuple(dirs)
    if getattr(ctx, "snapshot", None) is not None and getattr(ctx, "lib_dirs", None) == key:
        return ctx.snapshot
    snap = load_library(dirs)
    ctx.snapshot, ctx.lib, ctx.lib_dirs = snap, snap.lib, key
    return snap


def update_dir(dir_path: str, progress=None):
    conn = open_db(Path(dir_path) / "fontindex.db")
    try:
        return update_index(Path(dir_path), conn, progress=progress)
    finally:
        conn.close()


def read_failures(ctx) -> dict:
    """各目录 bad 清单 + 跨目录跳过（旧版本）。mode=ro 只读。"""
    out: dict[str, list[tuple[str, str]]] = {}
    for d in ctx.settings.get("font_dirs"):
        dbf = Path(d) / "fontindex.db"
        if not dbf.exists():
            continue
        try:
            conn = sqlite3.connect(dbf.resolve().as_uri() + "?mode=ro", uri=True)
            rows = conn.execute("SELECT rel_path, reason FROM bad").fetchall()
            conn.close()
        except sqlite3.Error:
            continue
        if rows:
            out[d] = rows
    skipped = ctx.snapshot.lib.skipped() if getattr(ctx, "snapshot", None) else []
    return {"bad": out, "skipped": skipped}


def retry_bad(ctx, dir_path: str, rel: str):
    """手动重试：移出 bad 清单后增量更新（size/mtime 未变的也会强制重解析）。"""
    conn = open_db(Path(dir_path) / "fontindex.db")
    conn.execute("DELETE FROM bad WHERE rel_path=?", (rel,))
    conn.commit()
    conn.close()
    return update_dir(dir_path)

def pull_chunk(searcher: FontSearcher, cancel: dict | None = None,
               chunk: int = SEARCH_CHUNK):
    """推进搜索器至 批满/暂停/耗尽/目标渲染失败/取消。
    返回 (candidates, state)；state ∈ running/paused/exhausted/
    target_error/cancelled。命中语义（暂停累计 hit_count、target_error
    懒解析一次）全部由 searcher 承担，此层不解读。"""
    cands = []
    state = "running"
    while len(cands) < chunk:
        if cancel is not None and cancel.get("cancelled"):
            state = "cancelled"
            break
        c = searcher.next()
        if c is None:
            state = ("target_error" if searcher.target_error
                     else "paused" if searcher.paused else "exhausted")
            break
        cands.append(c)
    return cands, state


def render_preview(entity, text: str):
    """预览渲染条：不走全局相似度缓存（键按 text_version），GUI 自带
    （指纹，文字）LRU。缺字跳过，绝不渲染 tofu。"""
    return render_bundle(Path(entity.dir_path) / entity.rel_path,
                         entity.face_index, text)


@dataclass
class ApplyFileResult:
    path: str
    out_path: str | None = None
    changed_lines: int = 0
    error: str | None = None


def apply_to_file(path: str, mapping: dict[str, str], suffix: str,
                  overwrite: bool) -> ApplyFileResult:
    src = Path(path)
    try:
        if overwrite:
            r = rewrite_file(src, mapping, overwrite_src=True)
            return ApplyFileResult(path, str(r.out_path), len(r.changed_lines))
        out = src.with_name(src.stem + suffix)
        r = rewrite_file(src, mapping, out=out)
        return ApplyFileResult(path, str(out), len(r.changed_lines))
    except Exception as e:  # noqa: BLE001  单文件失败不拖垮整批
        return ApplyFileResult(path, error=f"{type(e).__name__}: {e}")


def cache_stats(ctx) -> dict:
    return {"render_bytes": ctx.cache.render_bytes,
            "pair_count": ctx.cache.pair_count, "path": str(ctx.cache.path)}


def cache_clear(ctx) -> None:
    ctx.cache.clear_all()


def cache_move(ctx, new_path: str, migrate: bool) -> str:
    p = Path(new_path)
    if migrate and p.exists():
        p.unlink()                  # 覆盖既有文件：页面已弹确认框
    ctx.cache.move_to(p, migrate=migrate)
    ctx.settings.set("cache_path", str(p))
    return str(p)
```

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

from ass_font_substitute.gui.jobs import DirStatus


@dataclass
class DirRow:
    path: str
    state: str = "stale"        # ok / stale / indexing / offline
    added: int = 0
    changed: int = 0

    @property
    def badge(self) -> str:
        if self.state == "indexing":
            return "索引中"
        if self.state == "offline":
            return "不可用"
        if self.state == "stale":
            return f"需更新：新增 {self.added} · 修改 {self.changed}"
        return "已更新"


class LibraryVM:
    """字体库管理页状态机（零 Qt）。目录增删持久化进 settings.font_dirs。"""

    def __init__(self, settings):
        self._settings = settings
        self.dirs = [DirRow(p) for p in settings.get("font_dirs")]
        self.stats: dict = {}
        self.overlaps: list[str] = []
        self.failed_dirs: list[str] = []

    def add_dir(self, path: str) -> bool:
        p = str(Path(path))
        if any(Path(d.path) == Path(p) for d in self.dirs):
            return False
        self.dirs.append(DirRow(p))
        self.persist_dirs()
        return True

    def remove_dir(self, path: str) -> None:
        self.dirs = [d for d in self.dirs if Path(d.path) != Path(path)]
        self.persist_dirs()

    def refresh(self, statuses: list[DirStatus], snapshot_stats: dict,
                overlaps: list[str], offline_dirs: list[str]) -> None:
        by = {s.path: s for s in statuses}
        offline_set = {str(Path(p)) for p in offline_dirs}
        for d in self.dirs:
            if d.state == "indexing":
                continue                        # 更新进行中不被探测结果覆盖
            if d.path in offline_set:
                d.state, d.added, d.changed = "offline", 0, 0
                continue
            s = by.get(d.path)
            if s is None or s.state in ("missing", "offline"):
                d.state, d.added, d.changed = "offline", 0, 0
            else:
                d.state, d.added, d.changed = s.state, s.added, s.changed
        self.stats = snapshot_stats
        self.overlaps = list(overlaps)
        self.failed_dirs = [d.path for d in self.dirs if d.state == "offline"]

    def mark_indexing(self, path: str) -> None:
        d = self._row(path)
        if d:
            d.state = "indexing"

    def mark_done(self, path: str) -> None:
        d = self._row(path)
        if d:
            d.state = "ok"

    def persist_dirs(self) -> None:
        self._settings.set("font_dirs", [d.path for d in self.dirs])

    def _row(self, path: str) -> DirRow | None:
        return next((d for d in self.dirs if Path(d.path) == Path(path)), None)
```

```python
class IndexWorker(QThread):
    """逐目录跑 update_dir；progress 回调转发为信号，cancel() 经标志
    同时作用于回调返回值（停止当前目录剩余文件）与目录间循环。"""

    progress = Signal(str, int, int, object)     # (dir, done, total, UpdateResult)
    finished_dir = Signal(str, object)           # (dir, UpdateResult)
    failed = Signal(str, str)

    def __init__(self, dirs, parent=None):
        super().__init__(parent)
        self._dirs = list(dirs)
        self._cancel = False

    def cancel(self):
        self._cancel = True

    def run(self):
        from ass_font_substitute.gui.jobs import update_dir
        for d in self._dirs:
            if self._cancel:
                break
            try:
                r = update_dir(
                    d, progress=lambda done, total, res, d=d:
                    self._on_progress(d, done, total, res))
                self.finished_dir.emit(d, r)
            except Exception as e:  # noqa: BLE001
                self.failed.emit(d, f"{type(e).__name__}: {e}")

    def _on_progress(self, d, done, total, res) -> bool:
        if self._cancel:
            return False
        self.progress.emit(d, done, total, res)
        return True
```

```python
from __future__ import annotations

from collections import OrderedDict

from ass_font_substitute.core.fontindex.library import FontEntity

CATEGORY_ORDER = ["黑", "宋", "圆", "楷", "仿", "未分类"]
SCRIPT_ORDER = ["简", "繁", "简繁"]
TILE_CACHE_MAX = 2000


def weight_tag(w: int) -> str:
    return f"W{w // 100 * 100}"


class PreviewVM:
    """字体库预览页与「从字体库选择」对话框共享的筛选 + 预览 tile LRU（零 Qt）。"""

    def __init__(self):
        self.entities: list[FontEntity] = []
        self.name_query = ""
        self.category: str | None = None
        self.script: str | None = None
        self.weight: int | None = None
        self._tiles: OrderedDict = OrderedDict()

    def set_entities(self, entities) -> None:
        self.entities = list(entities)

    def weight_buckets(self) -> list[int]:
        return sorted({e.weight // 100 * 100 for e in self.entities})

    def categories(self) -> list[str]:
        present = {e.category for e in self.entities}
        return [c for c in CATEGORY_ORDER if c in present]

    def scripts(self) -> list[str]:
        present = {e.script for e in self.entities}
        return [s for s in SCRIPT_ORDER if s in present]

    def filtered(self) -> list[FontEntity]:
        q = self.name_query.strip().casefold()
        out = []
        for e in self.entities:
            if q and not any(q in a.casefold() for a in e.aliases):
                continue
            if self.category and e.category != self.category:
                continue
            if self.script and e.script != self.script:
                continue
            if self.weight is not None and e.weight // 100 * 100 != self.weight:
                continue
            out.append(e)
        return out

    # ---- tile LRU：键 (fingerprint, 预览文字) ----
    def get_tiles(self, key):
        t = self._tiles.pop(key, None)
        if t is not None:
            self._tiles[key] = t
        return t

    def put_tiles(self, key, tiles) -> None:
        self._tiles[key] = tiles
        self._tiles.move_to_end(key)
        while len(self._tiles) > TILE_CACHE_MAX:
            self._tiles.popitem(last=False)

    def invalidate(self) -> None:
        self._tiles.clear()
```

```python
from __future__ import annotations

import numpy as np
from PySide6.QtGui import QImage, QPixmap
from qfluentwidgets import CaptionLabel


def tiles_to_pixmap(tiles: list[np.ndarray]) -> QPixmap:
    """48×48 灰度 tile 横排成条；空列表返回空 QPixmap。"""
    if not tiles:
        return QPixmap()
    row = np.concatenate([np.pad(t, ((0, 0), (0, 1))) for t in tiles], axis=1)
    h, w = row.shape
    img = QImage(row.data, w, h, w, QImage.Format_Grayscale8)
    return QPixmap.fromImage(img)


class PreviewBar(CaptionLabel):
    """单行预览条：占位 → 异步渲染 → QPixmap。"""

    def __init__(self, worker, pvm, entity, text, tag, height=28, parent=None):
        super().__init__("…", parent)
        self.setFixedHeight(height)
        self._worker, self._pvm = worker, pvm
        self._entity, self._text, self._tag = entity, text, tag
        self._render()

    def _render(self):
        if not self._text.strip() or self._entity is None:
            self.setText(self._entity.display_name if self._entity else "")
            return
        key = (self._entity.fingerprint, self._text)
        cached = self._pvm.get_tiles(key)
        if cached is not None:
            self._apply(cached)
            return
        self._worker.result.connect(self._on_result)
        self._worker.submit(self._tag, _render_job, self._entity, self._text)

    def _on_result(self, tag, payload):
        if tag != self._tag:
            return
        self._worker.result.disconnect(self._on_result)
        self._pvm.put_tiles((self._entity.fingerprint, self._text), payload)
        self._apply(payload)

    def _apply(self, tiles):
        pm = tiles_to_pixmap(tiles)
        if pm.isNull():
            self.setText("(无可渲染字形)")
            return
        self.setPixmap(pm.scaledToHeight(self.height(),
                                         __import__("PySide6.QtCore", fromlist=["Qt"]).SmoothTransformation))


def _render_job(ctx, entity, text):
    from ass_font_substitute.gui.jobs import render_preview
    return render_preview(entity, text)
```

**注意**：`tiles_to_pixmap` 中 `np.pad(t, ((0,0),(0,1)))` 在 tile 间加 1px 间隔；`row.data` 要求内存连续（concatenate 结果天然连续）。`scaledToHeight` 的平滑转换 import 放文件头部：`from PySide6.QtCore import Qt`，用 `Qt.SmoothTransformation`（执行者按此清理上述动态 import）。

`_reco_job`：

```python
def _reco_job(ctx, chars, cov_threshold, direction):
    """target=None：无暂停语义一次搜穿整池（元数据 + 覆盖率，~15µs/候选）。"""
    s = make_searcher(ctx, target=None, chars=set(chars),
                      threshold=cov_threshold, n_pause=10 ** 9,
                      min_sim=0.0, size_range=(0.0, 10.0), direction=direction)
    cands, _ = pull_chunk(s, chunk=10 ** 9)
    fp2e = {e.fingerprint: e for e in ctx.lib.entities()}
    return [fp2e[c.fingerprint] for c in cands]
```

注意：`FontSearcher.__init__` 校验 `size_range[0] <= size_range[1]`，(0.0, 10.0) 合法；`min_sim=0.0` 让 target=None 分支不受影响（该分支本就不评相似度）。

```python
           "replace": "需替换", "missing": "库中没有"}


def weight_tag(w: int) -> str:
    return f"W{w // 100 * 100}"


@dataclass
class OptionItem:
    key: str
    display_name: str
    coverage: float | None = None
    similarity: float | None = None
    size_ratio: float | None = None
    meta: list[str] = field(default_factory=list)     # 简/繁/简繁 · W400 · 黑
    badges: list[str] = field(default_factory=list)   # 孪生对应/上次使用/原字体/不改动
    source: str = "search"                            # original/twin/search/last/pick
    n_chars: int | None = None
    entity: FontEntity | None = None                  # KEEP 行为 None


@dataclass
class Card:
    norm_name: str
    raw_name: str
    kind: str                                          # zero/keep/twin/replace/missing
    style_count: int = 0
    inline_count: int = 0
    n_chars: int = 0
    variants: set[str] = field(default_factory=set)
    missing_preview: list[str] = field(default_factory=list)   # 需替换卡 tofu 预览
    options: list[OptionItem] = field(default_factory=list)
    checked_key: str | None = None
    searching: bool = False
    search_state: str = "idle"                         # idle/paused/exhausted/target_error
    min_sim: float | None = None
    searcher: FontSearcher | None = None               # 仅 UI 线程读写；worker 只借用

    @property
    def header(self) -> str:
        return f"{self.style_count} 样式 · {self.inline_count} 行内 · {self.n_chars} 字"

    @property
    def status_text(self) -> str:
        return _STATUS[self.kind]

    @property
    def replaced(self) -> bool:
        """勾选了实体字体（将改写名称）。KEEP / 未勾 = False。"""
        return self.checked_key is not None and self.checked_key != KEEP

    def option(self, key: str) -> OptionItem | None:
        return next((o for o in self.options if o.key == key), None)


class SubsVM:
    """处理字幕页状态机（零 Qt）。跨线程约定：worker 只读 vm 公开字段并在
    回调里交还数据，一切变更走 UI 线程的 vm 方法。"""

    def __init__(self, settings, decisions_db: Path):
        self._settings = settings
        self.decisions_db = Path(decisions_db)
        self.direction = "t2s"
        self.files: list[FileParse] = []
        self.cards: list[Card] = []
        self.chars_by_norm: dict[str, set[str]] = {}
        self.ent_by_fp: dict[str, FontEntity] = {}

    # ---- 文件层 ----
    def add_files(self, parses: list[FileParse]) -> None:
        by = {f.path: f for f in self.files}
        for p in parses:
            by[p.path] = p
        self.files = list(by.values())

    def set_checked(self, path: str, checked: bool) -> None:
        for f in self.files:
            if f.path == path:
                f.checked = checked                    # FileParse 动态属性（见下）
                return
        raise KeyError(path)

    def checked_parses(self) -> list[FileParse]:
        return [f for f in self.files
                if getattr(f, "checked", True) and f.error is None]

    def merged_usages(self) -> list[FontUsage]:
        return merge_usages([f.usages for f in self.checked_parses()])
```

```python
    def make_card(self, usage: FontUsage, cls: UsageClass) -> Card:
        card = Card(norm_name=usage.norm_name, raw_name=usage.raw_name,
                    kind=cls.kind, style_count=usage.style_count,
                    inline_count=usage.inline_count, n_chars=len(usage.chars),
                    variants=set(usage.variants))
        self.chars_by_norm[usage.norm_name] = set(usage.chars)
        if cls.kind in ("zero", "missing"):
            card.options = [OptionItem(KEEP, "保留原字体名", badges=["不改动"])]
            return card
        opts = [self._entity_option(e, usage) for e in cls.entities]
        opts.sort(key=lambda o: -(o.coverage or 0.0))
        if cls.kind == "twin":
            twin = cls.twin_hit
            tcov, _ = coverage_detail(twin.cmap, usage.chars)
            twin_opt = OptionItem(twin.fingerprint, twin.display_name, coverage=tcov,
                                  meta=self._meta(twin), badges=["孪生对应"],
                                  source="twin", entity=twin)
            # 用户拍板 2026-10-23：孪生卡补列在库原字体（繁版可手选保留）
            card.options = [twin_opt] + [o for o in opts
                                         if o.key != twin.fingerprint]
        else:
            card.options = opts
        if cls.kind == "replace":
            card.missing_preview = list(cls.missing_chars)[:12]
        card.min_sim = float(self._settings.get("min_sim"))
        return card

    def _entity_option(self, e: FontEntity, usage: FontUsage) -> OptionItem:
        cov, _ = coverage_detail(e.cmap, usage.chars)
        is_orig = usage.norm_name in {normalize_name(a) for a in e.aliases}
        return OptionItem(e.fingerprint, e.display_name, coverage=cov,
                          meta=self._meta(e),
```

```python
    # ---- 搜索层 ----
    def attach_searcher(self, card: Card, searcher: FontSearcher) -> None:
        card.searcher = searcher
        card.searching = True
        card.search_state = "idle"

    def add_candidates(self, card: Card, candidates) -> list[OptionItem]:
        added = []
        for c in candidates:
            if card.option(c.fingerprint) is not None:
                continue
            o = OptionItem(c.fingerprint, c.display_name, coverage=c.coverage,
                           similarity=c.similarity, size_ratio=c.size_ratio,
                           meta=self._meta(self.ent_by_fp[c.fingerprint])
                           if c.fingerprint in self.ent_by_fp else [],
                           source="search", n_chars=c.n_chars,
                           entity=self.ent_by_fp.get(c.fingerprint))
            card.options.append(o)
            added.append(o)
        return added

    def finish_search(self, card: Card, searcher: FontSearcher) -> None:
        """终态：target_error（原字体无法渲染）> paused（可继续）>
        exhausted（无更多相似字体）。"""
        card.searching = False
        if searcher.target_error:
            card.search_state = "target_error"
        elif searcher.paused:
            card.search_state = "paused"
        else:
            card.search_state = "exhausted"

    def relax_min_sim(self, card: Card) -> float | None:
        """放宽重算门槛：降 10 个百分点，下限 50%。新门槛入 card.min_sim，
        由页面用新值建**新**搜索器（hit 计数清零）；旧结果保留——被拒 pair
        已入缓存，重算零渲染。已达下限返回 None。"""
        cur = card.min_sim if card.min_sim is not None \
            else float(self._settings.get("min_sim"))
        nxt = round(max(SIM_FLOOR, cur - RELAX_STEP), 2)
        if nxt >= cur:
            return None
        card.min_sim = nxt
        return nxt

    def _search_target(self, card: Card) -> FontEntity | None:
        """相似度基准 = 字幕原字体对应的在库实体（优先原名命中，否则覆盖
        最高者）；孪生体不作 target（已在选项里，指纹会经池排除逻辑互斥）。"""
        chars = self.chars_by_norm.get(card.norm_name, set())
        ents = [o.entity for o in card.options if o.entity is not None
                and "孪生对应" not in o.badges]
        if not ents:
            return None
        orig = next((e for e in ents
                     if card.norm_name in {normalize_name(a) for a in e.aliases}),
                    None)
        if orig is not None:
            return orig
        return max(ents, key=lambda e: e.cmap.hit_ratio(chars)) if chars else ents[0]

    def attach_pick(self, card: Card, entity: FontEntity) -> OptionItem:
        chars = self.chars_by_norm.get(card.norm_name, set())
        cov, _ = coverage_detail(entity.cmap, chars)
        o = card.option(entity.fingerprint)
        if o is None:
            o = OptionItem(entity.fingerprint, entity.display_name, coverage=cov,
                           meta=self._meta(entity), source="pick", entity=entity)
            card.options.append(o)
        card.checked_key = o.key
        return o
```

```python
    def build_mapping(self) -> dict[str, str]:
        """raw 拼写全集 → 选中实体名。勾选原字体行（source=original）不改写。"""
        m: dict[str, str] = {}

            if not c.replaced:
                continue
            o = c.option(c.checked_key)
            for v in c.variants:
                m[v] = o.display_name
        return m

    def replacement_counts(self) -> tuple[int, int]:
        return sum(1 for c in self.cards if c.replaced), len(self.cards)

    def _out_path(self, path: str) -> Path:
        return Path(path).with_name(Path(path).stem + self._settings.get("save_suffix"))

    def pending_out_conflicts(self) -> list[str]:
        return [str(self._out_path(f.path)) for f in self.checked_parses()
                if not self._settings.get("overwrite_default")
                and self._out_path(f.path).exists()]

    def apply_files(self, overwrite_confirm: bool = False) -> list:
        """逐勾选文件写回（jobs.apply_to_file）。另存目标已存在且未确认 →
        该文件 error="exists" 跳过；覆盖原文件模式无冲突。"""
        from ass_font_substitute.gui.jobs import apply_to_file
```

```python
def test_full_flow(world):
    ctx, vm, lib, sub = world
    fp = parse_subtitle(str(sub))
    vm.add_files([fp])
    usages = vm.merged_usages()
    assert len(usages) == 1
    cls = classify_one(lib, usages[0], 0.95, "t2s")
    assert cls.kind == "keep"                       # 测试黑体全覆盖 → 保留
    card = vm.make_card(usages[0], cls)
    vm.apply_default_check(card)
    vm.cards = [card]
    # 改选测试宋体 → 相似度搜索可跑通（有 target、有交集）
    target = vm._search_target(card)
    s = make_searcher(ctx, target=target, chars=set("永中木的水好"),
                      threshold=0.95, n_pause=5, min_sim=0.10,
                      size_range=(0.5, 2.0), direction="t2s")
    cands, state = pull_chunk(s)
    vm.add_candidates(card, cands)
    vm.finish_search(card, s)
    assert state in ("paused", "exhausted")
    assert any(o.display_name == "测试宋体" for o in card.options)
    card.checked_key = next(o.key for o in card.options
                            if o.display_name == "测试宋体")
    assert vm.replacement_counts() == (1, 1)
    rs = vm.apply_files()
    assert rs[0].error is None
    out = sub.with_name("demo.substituted.ass")
    assert "测试宋体" in out.read_text("utf-8")
    assert "测试黑体" not in out.read_text("utf-8")
    vm.record_all()
    from ass_font_substitute.core.decision.store import latest_decision
    assert latest_decision(vm.decisions_db, "测试黑体", "t2s")[0] == \
        card.checked_key
```
